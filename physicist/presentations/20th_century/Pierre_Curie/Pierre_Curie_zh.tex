% --- 7. 居里对称性原理 ---
\newcommand{\principleslide}{\begin{frame}\plainbar\sectiontitle{居里对称性原理}{效应不能具有原因所没有的不对称}
\vspace{-0.35cm}
\begin{center}

  \begin{minipage}{11.7cm}

  {\fontsize{8.6}{11.5}\selectfont\color{crystalblue!88!black}物理效应不能具有其原因所不具有的不对称性——结果中的每一处不对称，都必然来自原因。}

}
\end{center}
\vspace{0.25cm}
\begin{center}
\footnotesize
\renewcommand{\arraystretch}{0.8}
\arrayrulecolor{nobelgold!65}
\begin{tabularx}{12.4cm}{>{\bfseries\scriptsize}m{3.0cm}|>{\scriptsize}X|>{\scriptsize}p{3.0cm}}
\toprule
\rowcolor{crystalblue}\multicolumn{1}{c|}{\footnotesize\color{white}\textbf{思想实验}} & \multicolumn{1}{c|}{\footnotesize\color{white}\textbf{推理}} & \multicolumn{1}{c}{\footnotesize\color{white}\textbf{结论}} \\
\midrule
\cellcolor{goldpanel}\textcolor{nobelgold!88!black}{零重力下的沙粒} & \cellcolor{fampanel}\textcolor{badgeFam!82!black}{随机混合的沙粒在零重力中没有不对称性（各向同性）——沙粒分布处处均匀} & \cellcolor{silverpanel}\textcolor{coverdark!70}{原因无不对称，结果亦无} \\
\midrule
