\newcommand{\spacesslide}{\begin{frame}\plainbar\sectiontitle{Sobolev 空间 $W^{k,p}$：现代 PDE 的语言}{弱导数 · 完备性 · 嵌入定理 —— 每一门 PDE 研究生课的起点}
\vspace{-0.4cm}\begin{center}\begin{tikzpicture}[item/.style={rounded corners=4pt, inner xsep=10pt, inner ysep=5pt, text width=12.4cm, font=\fontsize{7.0}{9.0}\selectfont, align=left}]
\node[item, fill=redpanel, draw=badgeSob!60] at (0,2.35) {\textbf{定义。}$W^{k,p}(\Omega)$ = 所有 $k$ 阶弱导数属于 $L^p(\Omega)$ 的函数空间，配以范数 $\|u\|_{W^{k,p}} = (\sum_{|\alpha|\leq k} \|D^\alpha u\|_{L^p}^p)^{1/p}$。Banach 空间; $p=2$ 时为 Hilbert 空间 $H^k$。};
\node[item, fill=goldpanel, draw=badgeDist!40] at (0,0.65) {\textbf{嵌入定理。}Sobolev 嵌入: $W^{k,p}(\mathbb{R}^n) \hookrightarrow L^q(\mathbb{R}^n)$ 当 $1/q = 1/p - k/n$。Rellich–Kondrachov 紧嵌入: 有界域上 $W^{1,p} \subset\subset L^q$。};
\node[item, fill=greenpanel, draw=badgeAtom!25] at (0,-0.85) {\textbf{应用。}椭圆 PDE 的正则性 (Lax–Milgram), 抛物 PDE 的能量估计, 双曲 PDE 的守恒律, Navier–Stokes 方程的弱解。没有 Sobolev 空间就没有现代 PDE。};
\node[draw=coveraccent!55, fill=graypanel, rounded corners=5pt, inner sep=6pt, text width=12.4cm, align=center] at (0,-2.1) {{\fontsize{8.0}{10.0}\selectfont\bfseries\color{coverdark!88} 分布理论 + Sobolev 空间 + 嵌入定理 = 现代 PDE 的语法三部曲。}}\end{tikzpicture}\end{center}\end{frame}}
