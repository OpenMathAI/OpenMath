\begin{tikzpicture}[
  item/.style={rounded corners=4pt, inner xsep=10pt, inner ysep=5pt, text width=12.4cm, font=\fontsize{7.0}{9.0}\selectfont, align=left}

  \node[item, fill=greenpanel, draw=badgeMorse!55] at (0,2.35)
    {\textbf{弱不等式：}$m_k \geq b_k$，其中 $m_k$ 是指标为 $k$ 的临界点数量，$b_k$ 是第 $k$ 个 Betti 数。Betti 数是流形的拓扑不变量（纯拓扑概念）——而临界点数量取决于函数的选择（分析/几何概念）。};
  \node[item, fill=bluepanel, draw=badgetopo!40] at (0,0.85)
    {\textbf{强不等式：}$\sum_{i=0}^k (-1)^{k-i} m_i \geq \sum_{i=0}^k (-1)^{k-i} b_i$。更强的交错不等式给出了拓扑与分析之间的精确约束。当 $k=n$ 时退化为 Poincare-Hopf 指标公式的推广。};
  \node[item, fill=bronzepanel, draw=badgecalc!25] at (0,-0.45)
    {\textbf{重要性。}(1) 拓扑信息可以从分析信息推导 (2) 为微分拓扑提供了基本工具 (3) Bott, Smale, Milnor 的突破都建立在这组不等式之上。};

    {\fontsize{8.0}{10.0}\selectfont\bfseries\color{coverdark!88} 在这组不等式之前，没有人想过可以用分析的工具来计算拓扑。}
