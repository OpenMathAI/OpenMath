\newcommand{\griffithsslide}{\personslide
  {Phillip Griffiths}{菲利普·格里菲斯}
  {images/griffiths.jpg}{Wikimedia Commons}
  {2014（76岁）}{1938– }{美国 USA}{普林斯顿高等研究院}

\newcommand{\kashiwaraslide}{\personslide
  {Masaki Kashiwara}{柏原正树}
  {images/kashiwara.jpg}{日本学士院 / Wikimedia}
  {2018（71岁）}{1947– }{日本 Japan}{京都大学数理解析研究所}

\newcommand{\mazurslide}{\personslide
  {Barry Mazur}{巴里·梅休}
  {images/mazur.jpg}{Wikimedia Commons}
  {2022（85岁）}{1937– }{美国 USA}{哈佛大学}
