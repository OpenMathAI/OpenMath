\newcommand{\bernardsilvermanslide}{\personslide
  {Bernard Silverman}{伯纳德·西尔弗曼}
  {images/silverman.jpg}{Wikipedia}
  {1991}{1952–}{英国}{University of Oxford}
  {研究方向：密度估计与非参数回归。密度估计、非参数回归、函数型数据分析。他的经典著作定义了密度估计这一方向。}}

\newcommand{\nancyreidslide}{\personslide
  {Nancy Reid}{南希·里德 \enspace·\enspace 首位女性得主}
  {images/reid.jpg}{Wikipedia}
  {1992}{1952–}{加拿大}{University of Toronto}
  {研究方向：渐近理论与复合似然。复合似然、似然高阶渐近、鞍点近似。她是加拿大统计学的旗帜人物。}}

\newcommand{\winghungwongslide}{\personslide
  {Wing Hung Wong}{黄永康}
  {images/winghungwong.jpg}{W.H.W.}
  {1993}{1952–}{美国（生于香港）}{Stanford University}
  {研究方向：MCMC 与贝叶斯计算。MCMC 方法、吉布斯抽样理论、计算生物学。他为贝叶斯计算的实用化作出开创性贡献。}}

\newcommand{\davidldonohoslide}{\personslide
  {David L. Donoho}{戴维·多诺霍}
  {images/donoho.jpg}{Wikipedia}
  {1994}{1957–}{美国}{Stanford University}
  {研究方向：小波与压缩感知。小波分析、压缩感知、高维统计推断。他是压缩感知的奠基人之一，2018 年获香农奖。}}

\newcommand{\iainmjohnstoneslide}{\personslide
  {Iain M. Johnstone}{伊恩·约翰斯通}
  {images/johnstone.jpg}{Wikipedia}
  {1995}{1956–}{美国（生于加拿大）}{Stanford University}
  {研究方向：随机矩阵与高维统计。随机矩阵理论、小波阈值去噪、稀疏主成分。他在高维统计的理论化上影响深远。}}

\newcommand{\robertjtibshiranislide}{\personslide
  {Robert J. Tibshirani}{罗伯特·蒂布希拉尼}
  {images/tibshirani.jpg}{Wikipedia}
  {1996}{1956–}{美国/加拿大}{Stanford University}
  {研究方向：LASSO 与统计学习。LASSO、Bootstrap、交叉验证。他与 Hastie、Friedman 合著的 ESL 是现代统计学习经典。}}

\newcommand{\kathrynroederslide}{\personslide
  {Kathryn Roeder}{凯瑟琳·罗德}
  {images/kathrynroeder.jpg}{K.R.}
  {1997}{1960–}{美国}{Carnegie Mellon University}
  {研究方向：统计遗传学与多重检验。统计遗传学、混合物模型、法医统计。她在疾病基因定位的统计方法上贡献突出。}}

\newcommand{\pascalmassartslide}{\personslide
  {Pascal Massart}{帕斯卡尔·马萨尔 \enspace·\enspace 首位法国得主}
  {images/pascalmassart.jpg}{P.M.}
  {1998}{1958–}{法国}{Université de Paris-Sud}
  {研究方向：浓度不等式与模型选择。浓度不等式、经验过程、模型选择理论。他与 Bousquet 的工作重塑了非渐近统计学。}}

\newcommand{\larryawassermanslide}{\personslide
  {Larry A. Wasserman}{拉里·瓦瑟曼}
  {images/wasserman.png}{Wikipedia}
  {1999}{1959–}{美国/加拿大}{Carnegie Mellon University}
  {研究方向：非参数推断。非参数推断、置信带方法、统计学习。他的 All of Statistics 是流传最广的统计入门之一。}}

\newcommand{\jianqingfanslide}{\personslide
  {Jianqing Fan}{范剑青 \enspace·\enspace 首位华人·普林斯顿}
  {images/fan.jpg}{Wikipedia}
  {2000}{1962–}{美国（生于中国）}{Princeton University}
  {研究方向：非参数与高维统计。非参数模型、变系数模型、高维统计学习。他是华人统计学家的杰出代表。}}
