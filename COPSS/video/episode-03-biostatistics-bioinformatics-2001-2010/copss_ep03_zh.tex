\newcommand{\xiaolimengslide}{\personslide
  {Xiao-Li Meng}{孟晓犁}
  {images/xiaolimeng.jpg}{X.M.}
  {2001}{1963–}{美国（生于中国）}{Harvard University}
  {研究方向：贝叶斯统计与计算。EM 算法理论、MCMC 收敛诊断、缺失数据。他是哈佛统计系创系主任。}}

\newcommand{\junliuslide}{\personslide
  {Jun Liu}{刘军}
  {images/junliu.jpg}{J.L.}
  {2002}{1965–}{美国（生于中国）}{Harvard University}
  {研究方向：MCMC 与计算生物学。MCMC 方法、序贯蒙特卡洛、系统生物学。他是贝叶斯计算与计算生物学交叉的领军者。}}

\newcommand{\andrewgelmanslide}{\personslide
  {Andrew Gelman}{安德鲁·格尔曼}
  {images/gelman.jpg}{Wikipedia}
  {2003}{1965–}{美国}{Columbia University}
  {研究方向：多层次贝叶斯模型。多层次贝叶斯模型、Stan 概率编程、后验预测检查。BDA 经典教材影响一代统计学家。}}

\newcommand{\michaelanewtonslide}{\personslide
  {Michael A. Newton}{迈克尔·牛顿}
  {images/michaelanewton.jpg}{M.N.}
  {2004}{1960–}{美国}{University of Wisconsin}
  {研究方向：计算生物学。计算生物学、基因表达数据分析、贝叶斯推断。他在基因组统计方法上持续深耕。}}

\newcommand{\markjvanderlaanslide}{\personslide
  {Mark J. van der Laan}{马克·范德兰}
  {images/markjvanderlaan.jpg}{M.L.}
  {2005}{1967–}{美国（生于荷兰）}{UC Berkeley}
  {研究方向：目标学习与因果推断。目标极大似然估计（TMLE）、超级学习者、半参数因果推断。他为观察性研究提供严谨推断框架。}}

\newcommand{\xihonglinslide}{\personslide
  {Xihong Lin}{林希虹}
  {images/lin.jpg}{Wikipedia}
  {2006}{1966–}{美国（生于中国）}{Harvard University}
  {研究方向：统计遗传学。基因-环境交互、全基因组关联分析（GWAS）。她是华人女性统计学家典范。}}

\newcommand{\jeffrosenthalslide}{\personslide
  {Jeff Rosenthal}{杰夫·罗森塔尔}
  {images/rosenthal.jpg}{Wikipedia}
  {2007}{1967–}{加拿大}{University of Toronto}
  {研究方向：MCMC 收敛理论。MCMC 收敛理论、随机过程模拟。他的随机游走 Metropolis 收敛分析被广泛应用。}}

\newcommand{\ttonycaislide}{\personslide
  {T. Tony Cai}{蔡天文}
  {images/cai.jpg}{Wikipedia}
  {2008}{1967–}{美国（生于中国）}{University of Pennsylvania}
  {研究方向：高维统计推断。高维协方差矩阵估计、非参数函数估计极小极大最优、多重检验。他是华人高维统计的代表。}}

\newcommand{\rafaelirizarryslide}{\personslide
  {Rafael Irizarry}{拉斐尔·伊里萨里}
  {images/rafaelirizarry.jpg}{R.I.}
  {2009}{1972–}{美国（波多黎各）}{Harvard University}
  {研究方向：基因组数据分析。基因组数据标准化、微阵列方法、数据可视化。他的工具推动了生物信息学的标准化。}}

\newcommand{\daviddunsonslide}{\personslide
  {David Dunson}{戴维·邓森}
  {images/daviddunson.jpg}{D.D.}
  {2010}{1972–}{美国}{Duke University}
  {研究方向：贝叶斯非参数。贝叶斯非参数、高维潜变量模型、函数型数据先验。他在贝叶斯非参数的现代化上影响深远。}}
