\newcommand{\nilanjanchatterjeeslide}{\personslide
  {Nilanjan Chatterjee}{尼兰詹·查特吉 \enspace·\enspace 首位印度裔得主}
  {images/chatterjee.png}{Wikipedia}
  {2011（39岁）}{1972–}{美国（生于印度）}{Johns Hopkins University}
  {研究方向：统计遗传学与流行病学。多基因风险评分、癌症风险建模、流行病学方法。他在疾病风险预测上贡献突出。\\\textbf{主要荣誉}：Spiegelman 奖 2010；Snedecor 奖 2011}}

\newcommand{\samuelkouslide}{\personslide
  {Samuel Kou}{寇星昌}
  {images/samuelkou.jpg}{S.K.}
  {2012（38岁）}{1974–}{美国（生于中国）}{Harvard University}
  {研究方向：随机过程与生物物理。单分子生物物理、随机过程与统计推断。他建立了单分子数据的统计推断框架。}}

\newcommand{\marcasuchardslide}{\personslide
  {Marc A. Suchard}{马克·苏查德}
  {images/marcasuchard.jpg}{M.S.}
  {2013（41岁）}{1972–}{美国}{UCLA}
  {研究方向：贝叶斯系统发生学。贝叶斯系统发生学、BEAST 软件、MCMC 计算。他的软件让系统发生学进入贝叶斯时代。}}

\newcommand{\martinjwainwrightslide}{\personslide
  {Martin J. Wainwright}{马丁·温赖特}
  {images/martinjwainwright.jpg}{M.W.}
  {2014（41岁）}{1973–}{美国/加拿大}{UC Berkeley}
  {研究方向：高维统计与图模型。高维统计、图模型、信息论与优化。他的 High-Dimensional Statistics 是该领域标准教材。\\\textbf{主要荣誉}：Sloan 学者 2005；Guggenheim 学者 2024}}

\newcommand{\johndstoreyslide}{\personslide
  {John D. Storey}{约翰·斯托里}
  {images/johndstorey.jpg}{J.S.}
  {2015（37岁）}{1978–}{美国}{Princeton University}
  {研究方向：多重检验与 q-value。q-value 框架、FDR 控制、统计基因组学。他的方法改变了组学数据分析。\\\textbf{主要荣誉}：Spiegelman 奖 2015}}

\newcommand{\nicolaimeinshausenslide}{\personslide
  {Nicolai Meinshausen}{尼古拉·迈因斯豪森 \enspace·\enspace 首位瑞士得主}
  {images/nicolaimeinshausen.jpg}{N.M.}
  {2016（38岁）}{1978–}{瑞士/德国}{ETH Zürich}
  {研究方向：高维统计与因果。高维变量选择、不变因果预测、随机森林稳定性。他连接了高维统计与因果推断。}}

\newcommand{\tylerjvanderweeleslide}{\personslide
  {Tyler J. VanderWeele}{泰勒·范德维勒}
  {images/vanderweele.jpg}{Wikipedia}
  {2017（40岁）}{1977–}{美国}{Harvard University}
  {研究方向：因果中介分析。因果中介分析、直接与间接效应分解、流行病学方法。他是现代因果推断方法论的代表人物，也研究宗教与健康的关系。\\\textbf{主要荣誉}：Spiegelman 奖 2014}}

\newcommand{\richardjsamworthslide}{\personslide
  {Richard J. Samworth}{理查德·萨姆沃思}
  {images/samworth.jpg}{Wikipedia}
  {2018（40岁）}{1978–}{英国}{University of Cambridge}
  {研究方向：高维非参数估计。高维非参数估计、变点检测、聚类与分类。他在理论统计与数据科学的交叉上成果丰硕。}}

\newcommand{\hadleywickhamslide}{\personslide

\newcommand{\rinafoygelbarberslide}{\personslide
