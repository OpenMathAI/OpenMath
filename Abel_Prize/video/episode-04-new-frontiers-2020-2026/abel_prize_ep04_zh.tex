\newcommand{\margulisslide}{\personslide

\newcommand{\lovaszslide}{\personslide
  {László Lovász}{拉斯洛·洛瓦斯\enspace·\enspace 与 Wigderson 共享}

  {2021}{1948– }{匈牙利 Hungary}{厄特沃什·罗兰大学}

\newcommand{\wigdersonslide}{\personslide
  {Avi Wigderson}{阿维·维格德森\enspace·\enspace 与 Lovász 共享}

\newcommand{\sullivanslide}{\personslide
  {Dennis Sullivan}{丹尼斯·苏利文}
  {images/sullivan.jpg}{Wikimedia Commons}

\newcommand{\caffarellislide}{\personslide

  {2023}{1948– }{阿根廷／美国}{德克萨斯大学奥斯汀分校}

\newcommand{\talagrandslide}{\personslide

\newcommand{\kashiwaraslide}{\personslide

\newcommand{\faltingsslide}{\personslide

  {2026}{1954– }{德国 Germany}{马克斯·普朗克数学研究所}
