\begin{frame}

\vspace{-0.4cm}
\begin{center}
\begin{tikzpicture}[
  item/.style={rounded corners=4pt, inner xsep=10pt, inner ysep=5pt, text width=12.4cm, font=\fontsize{7.0}{9.0}\selectfont, align=left}

  \node[item, fill=redpanel, draw=badgeinteg!55] at (0,2.35)
    {\textbf{视角转换。}Riemann 积分按定义域（x 轴）切分——先竖后横。Lebesgue 积分按值域（y 轴）切分——先横后竖。在 y 轴上取水平切片，测量函数取值落在每个小区间内的 x 集合的"测度"，然后求和。};
  \node[item, fill=graypanel, draw=badgemeasure!40] at (0,0.85)
    {\textbf{简单函数 → 可测函数 → 一般函数。}Lebesgue 先为简单函数（只取有限多个值的可测函数）定义积分，再用上确界逼近为任意可测函数定义积分。这个"从简单到复杂"的构造方式成为现代分析的标准模板。};
  \node[item, fill=bluepanel, draw=badgefourier!35] at (0,-0.45)
    {\textbf{更强的收敛定理。}Lebesgue 积分的核心优势：控制收敛定理（Dominated Convergence Theorem）。在非常宽松的条件下即可交换极限和积分——这是 Riemann 积分永远做不到的。};
  \node[draw=coveraccent!55, fill=graypanel, rounded corners=5pt, inner sep=6pt, text width=12.4cm, align=center] at (0,-2.0) {
    {\fontsize{8.0}{10.0}\selectfont\bfseries\color{coverdark!88} Riemann 用竖线切蛋糕。Lebesgue 说：用横线切。就这么简单。}
  };
\end{tikzpicture}
\end{center}
\end{frame}

\begin{frame}
\plainbar
\sectiontitle{积分应该满足的六个条件}{从公理化视角看积分 —— 比 Hilbert 的几何公理更早的"公理化精神"}
\vspace{-0.4cm}
\begin{center}
\begin{tikzpicture}[
  item/.style={rounded corners=4pt, inner xsep=10pt, inner ysep=5pt, text width=12.4cm, font=\fontsize{7.0}{9.0}\selectfont, align=left}
]
  \node[item, fill=redpanel, draw=badgeinteg!50] at (0,2.35)
    {\textbf{1904 《积分教程》。}在著名的"Borel 文集"中，Lebesgue 提出了积分应该满足的六个理想条件。这不是具体的计算规则，而是一种公理化思考——积分是什么？一个好的积分理论需要具备哪些性质？};
  \node[item, fill=graypanel, draw=badgemeasure!35] at (0,0.85)
    {\textbf{关键条件六。}"如果函数序列 $f_n(x)$ 递增地趋向极限 $f(x)$，则 $f_n$ 的积分趋向 $f$ 的积分。"这是 Lebesgue 积分比 Riemann 积分更强大的核心原因：单调收敛定理。};
