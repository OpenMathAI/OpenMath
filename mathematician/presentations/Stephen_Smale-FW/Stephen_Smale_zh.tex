% ===== 04. 高维 Poincaré 猜想 =====
\newcommand{\poincareslide}{\begin{frame}\plainbar\sectiontitle{高维 Poincaré 猜想 (1961) — Fields 奖的核心}{Smale 证明了 n ≥ 5 时的广义 Poincaré 猜想——打破了维度壁垒}
\vspace{-0.4cm}\begin{center}\begin{tikzpicture}[item/.style={rounded corners=4pt, inner xsep=10pt, inner ysep=5pt, text width=12.4cm, font=\fontsize{7.0}{9.0}\selectfont, align=left}]
\node[item, fill=bluepanel, draw=badgeTop!60] at (0,2.55) {\textbf{Poincaré 猜想。}每一个单连通闭三维流形是否同胚于三维球面 $S^3$？Poincaré (1904) 提出的原始问题——困扰数学家近 60 年。低维 (3 和 4) 极其困难——Perelman 直到 2003 年才解决三维情形。};
\node[item, fill=goldpanel, draw=badgeDS!40] at (0,0.65) {\textbf{Smale 的突破 (1961)。}对于 $n \geq 5$ 维, Smale 证明了广义 Poincaré 猜想。关键工具: 自索引 Morse 函数——将 Morse 理论和柄体分解 (handle decomposition) 结合, 用梯度流消除不必要的临界点。};
\node[item, fill=redpanel, draw=badgeHorse!25] at (0,-1.05) {\textbf{h-协边定理 (1962)。}Smale 进一步证明了更强大的 h-协边定理——如果两个单连通流形之间的协边是同伦平凡的, 则它们微分同胚。这给出了五维单连通光滑流形的完整分类。};
\node[draw=coveraccent!55, fill=graypanel, rounded corners=5pt, inner sep=6pt, text width=12.4cm, align=center] at (0,-2.2) {{\fontsize{8.0}{10.0}\selectfont\bfseries\color{coverdark!88} Smale 证明了前人认为"过于困难"的高维情形——因为高维给了更多操作空间。}}\end{tikzpicture}\end{center}\end{frame}}

% ===== 05. 球面翻转 =====
\newcommand{\eversionslide}{\begin{frame}\plainbar\sectiontitle{球面翻转 (Sphere Eversion) — Smale 悖论}{不撕裂和不产生折痕的情况下把球面从里向外翻转——可能吗?}
\vspace{-0.4cm}\begin{center}\begin{tikzpicture}[item/.style={rounded corners=4pt, inner xsep=10pt, inner ysep=5pt, text width=12.4cm, font=\fontsize{7.0}{9.0}\selectfont, align=left}]
\node[item, fill=bluepanel, draw=badgeTop!55] at (0,2.55) {\textbf{问题。}能否通过一族浸入 (immersion) 将标准球面连续变形为其自身的"翻转"版本 (法向量处处反向)？直观上似乎不可能——但 Smale 证明这是可能的。};
\node[item, fill=goldpanel, draw=badgeDS!35] at (0,0.85) {\textbf{Smale 的证明 (1959)。}他研究了二维球面浸入到三维欧氏空间的问题。通过将浸入理论转化为 Stiefel 流形的代数拓扑问题, 他完全分类了浸入的正则同伦类。球面翻转是直接的推论。};
\node[item, fill=redpanel, draw=badgeHorse!25] at (0,-0.65) {\textbf{影响。}Hirsch–Smale 浸入理论后来被 Gromov 发展为 h-原理——这是微分拓扑中最重要的概念之一。球面翻转成为拓扑学可视化最著名的例子之一。};
\node[draw=coveraccent!55, fill=graypanel, rounded corners=6pt, inner sep=6pt, text width=12.4cm, align=center] at (0,-2.0) {{\fontsize{8.0}{10.0}\selectfont\bfseries\color{coverdark!88} "球面翻转"被证明存在——尽管没有人能在物理现实中想象它。这是拓扑学超越直觉的力量。}}\end{tikzpicture}\end{center}\end{frame}}

% ===== 06. 马蹄映射 =====
\newcommand{\horseshoeslide}{\begin{frame}\plainbar\sectiontitle{马蹄映射 (1967) — 混沌的拓扑学起源}{一个简单的几何构造揭示了确定性混沌的本质结构}
\vspace{-0.4cm}\begin{center}\begin{tikzpicture}[item/.style={rounded corners=4pt, inner xsep=10pt, inner ysep=5pt, text width=12.4cm, font=\fontsize{7.0}{9.0}\selectfont, align=left}]
\node[item, fill=redpanel, draw=badgeHorse!60] at (0,2.35) {\textbf{定义。}马蹄映射是一个正方形上的微分同胚: 先将正方形沿一个方向拉伸、沿另一个方向压缩, 然后弯折成马蹄形放回正方形。迭代后产生 Cantor 集上的符号动力系统。};
\node[item, fill=goldpanel, draw=badgeDS!40] at (0,0.65) {\textbf{数学意义。}马蹄映射具有无穷多个周期轨道、对初始条件的敏感依赖、以及稠密轨道——混沌的三个核心特征。Smale 用这个简单的模型证明了混沌是结构稳定 (structurally stable) 的。};
\node[item, fill=bluepanel, draw=badgeTop!25] at (0,-0.85) {\textbf{"可微动力系统" (1967)。}这篇发表在 Bull. AMS 上的 71 页综述论文彻底革新了动力系统理论。Smale 将拓扑学和动力系统融合, 建立了现代动力系统理论的框架——包括 Axiom A 和 $\Omega$-稳定性。};
\node[draw=coveraccent!55, fill=graypanel, rounded corners=5pt, inner sep=6pt, text width=12.4cm, align=center] at (0,-2.1) {{\fontsize{8.0}{10.0}\selectfont\bfseries\color{coverdark!88} 一个简单的马蹄——却包含了混沌数学的全部DNA。这是动力系统理论的"E = mc²"时刻。}}\end{tikzpicture}\end{center}\end{frame}}
