\newcommand{\hprincipleslide}{\begin{frame}\plainbar\sectiontitle{h-原理 (Homotopy Principle) — "柔性"几何的革命}{当拓扑学和几何学相遇——微分关系的同伦方法}
\vspace{-0.4cm}\begin{center}\begin{tikzpicture}[item/.style={rounded corners=4pt, inner xsep=10pt, inner ysep=5pt, text width=12.4cm, font=\fontsize{7.0}{9.0}\selectfont, align=left}]
\node[item, fill=goldpanel, draw=badgeHP!60] at (0,2.55) {\textbf{核心思想。}h-原理：对于某些微分关系 (如浸入条件), 从形式解 (仅在代数拓扑层面成立) 到真正解 (满足实际微分方程) 的障碍是"柔性的"——可以用纯同伦论方法克服。};
\node[item, fill=redpanel, draw=badgeHG!40] at (0,0.65) {\textbf{凸积分 (Convex Integration)。}Gromov 在 1973 年引入了凸积分技术——通过逐点微扰将形式解转化为真正解。这是 h-原理最核心的工具, 应用于浸入、嵌入、等距嵌入 (Nash–Kuiper) 等经典问题。};
\node[item, fill=bluepanel, draw=badgeSY!25] at (0,-1.05) {\textbf{影响。}h-原理统一了 Smale (球面翻转), Hirsch (浸入分类), Nash (等距嵌入) 的分散结果, 将它们纳入一个统一的哲学框架。1986 年专著《Partial Differential Relations》是领域圣经。};
\node[draw=coveraccent!55, fill=graypanel, rounded corners=5pt, inner sep=6pt, text width=12.4cm, align=center] at (0,-2.2) {{\fontsize{8.0}{10.0}\selectfont\bfseries\color{coverdark!88} h-原理说：如果拓扑上存在解, 那么几何上就存在解。这是微分拓扑中最深刻的洞察之一。}}\end{tikzpicture}\end{center}\end{frame}}

\newcommand{\hyperbolicslide}{\begin{frame}\plainbar\sectiontitle{双曲群 (1987) — 几何群论的诞生}{$\delta$-双曲空间 · 无限群中的"负曲率"——一个全新的数学世界}
\vspace{-0.4cm}\begin{center}\begin{tikzpicture}[item/.style={rounded corners=4pt, inner xsep=10pt, inner ysep=5pt, text width=12.4cm, font=\fontsize{7.0}{9.0}\selectfont, align=left}]
\node[item, fill=redpanel, draw=badgeHG!60] at (0,2.35) {\textbf{$\delta$-双曲空间。}Gromov 定义了测地度量空间的"双曲性"——存在常数 $\delta$ 使得任意测地三角形的每条边都在另外两条边的 $\delta$-邻域内 (瘦三角形条件)。这是负曲率的最本质抽象。};
\node[item, fill=goldpanel, draw=badgeHP!40] at (0,0.65) {\textbf{双曲群。}一个有限生成群是双曲的, 如果它的 Cayley 图 (配以字度量) 是一个 $\delta$-双曲空间。自由群、曲面群、大多数负曲率流形的基本群都是双曲群。};
\node[item, fill=bluepanel, draw=badgeSY!25] at (0,-0.85) {\textbf{革命性影响。}Gromov 的 1987 年论文《Hyperbolic Groups》彻底改变了无限群论——将几何概念引入纯代数领域。双曲群具有可解的字问题、无挠、有限表示——为群论带来了丰富的几何工具。};
\node[draw=coveraccent!55, fill=graypanel, rounded corners=5pt, inner sep=6pt, text width=12.4cm, align=center] at (0,-2.1) {{\fontsize{8.0}{10.0}\selectfont\bfseries\color{coverdark!88} 双曲群——Gromov 用几何学重写了群论的语言, 开创了"几何群论"这个全新的学科。}}\end{tikzpicture}\end{center}\end{frame}}
