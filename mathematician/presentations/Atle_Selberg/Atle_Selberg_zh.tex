\newcommand{\traceslide}{\begin{frame}\plainbar\sectiontitle{Selberg 迹公式 (1956)}{调和分析 $\longleftrightarrow$ 数论——20 世纪最深刻的公式之一}
\vspace{-0.4cm}\begin{center}\begin{tikzpicture}[item/.style={rounded corners=4pt, inner xsep=10pt, inner ysep=5pt, text width=12.4cm, font=\fontsize{7.0}{9.0}\selectfont, align=left}]
\node[item, fill=goldpanel, draw=badgeSF!60] at (0,2.55) {\textbf{公式的本质。}迹公式将 Laplace 算子的特征值（谱、分析侧）与闭测地线的长度（几何侧）联系起来：$\sum_n h(\lambda_n) = \sum_\gamma \text{length}(\gamma) \cdot g(\text{length}(\gamma))$。};
\node[item, fill=bluepanel, draw=badgeNT!40] at (0,0.65) {\textbf{应用。}数论：自守形式的谱分解，Riemann zeta 函数的零点分布。几何：曲面上测地线的长度谱。物理：量子混沌中的 Gutzwiller 迹公式就是 Selberg 迹公式的物理版本。};
\node[item, fill=greenpanel, draw=badgeSie!25] at (0,-1.05) {\textbf{Selberg $\zeta$ 函数。}与 Riemann zeta 函数类似，但在双曲曲面上定义：$\prod_{\gamma} \prod_{k=1}^\infty (1 - e^{-(s+k) \cdot \text{length}(\gamma)})$。满足函数方程，像 Riemann zeta 函数一样神秘。};
\node[draw=coveraccent!55, fill=graypanel, rounded corners=5pt, inner sep=6pt, text width=12.4cm, align=center] at (0,-2.2) {{\fontsize{8.0}{10.0}\selectfont\bfseries\color{coverdark!88} 一个公式，连接了数论、几何和物理——Selberg 迹公式是 20 世纪数学的统一力量。}}\end{tikzpicture}\end{center}\end{frame}}
