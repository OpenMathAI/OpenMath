\newcommand{\psislide}{\begin{frame}\plainbar\sectiontitle{伪微分算子 ($\Psi$DO) 与 Fourier 积分算子 (FIO)}{微局部分析——在相空间中研究 PDE}
\vspace{-0.4cm}\begin{center}\begin{tikzpicture}[item/.style={rounded corners=4pt, inner xsep=10pt, inner ysep=5pt, text width=12.4cm, font=\fontsize{7.0}{9.0}\selectfont, align=left}]
\node[item, fill=goldpanel, draw=badgePsi!60] at (0,2.35) {\textbf{伪微分算子 $\Psi$DO。}推广了微分算子和奇异积分算子：$Au(x) = \int e^{i x\cdot\xi} a(x,\xi) \hat{u}(\xi) d\xi$。符号演算——算子的乘积、伴随、逆都可以在符号层面操作。};
\node[item, fill=greenpanel, draw=badgeFIO!40] at (0,0.65) {\textbf{Fourier 积分算子 FIO (1971)。}更一般的相位函数：$Au(x) = \int e^{i \phi(x,\xi)} a(x,\xi) \hat{u}(\xi) d\xi$。FIO 描述了波动方程解的传播——奇异性的传播沿 Hamiltonian 流的双特征线。};
\node[item, fill=bluepanel, draw=badgeLPO!25] at (0,-0.85) {\textbf{微局部分析 (microlocal analysis)。}Hörmander 与 Sato (日本) 独立发展了微局部分析——在余切丛 $T^*M$ 上局部化。波前集 (wave front set) $\mathrm{WF}(u)$ 描述了分布 $u$ 的奇异性方向。};
\node[draw=coveraccent!55, fill=graypanel, rounded corners=5pt, inner sep=6pt, text width=12.4cm, align=center] at (0,-2.1) {{\fontsize{8.0}{10.0}\selectfont\bfseries\color{coverdark!88} $\Psi$DO + FIO + 微局部分析 = 现代 PDE 理论的三角核心。}}\end{tikzpicture}\end{center}\end{frame}}
