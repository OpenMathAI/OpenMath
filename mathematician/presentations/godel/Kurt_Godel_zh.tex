  \node[draw=coveraccent!55, fill=graypanel, rounded corners=5pt, inner sep=6pt, text width=12.4cm, align=center] at (0,-1.8) {
    {\fontsize{8.0}{10.0}\selectfont\bfseries\color{coverdark!88} 真理 > 证明。这是 Gödel 留给数学最深层的洞见。}

\vspace{-0.4cm}
\begin{center}
\begin{tikzpicture}[
  item/.style={rounded corners=4pt, inner xsep=10pt, inner ysep=5pt, text width=12.4cm, font=\fontsize{7.0}{9.0}\selectfont, align=left}

  \node[item, fill=goldpanel, draw=badgeset!55] at (0,2.35)
