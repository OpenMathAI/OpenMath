\begin{center}
\begin{tikzpicture}[
  item/.style={rounded corners=4pt,inner xsep=10pt,inner ysep=5pt,text width=12.4cm,font=\fontsize{7.0}{9.0}\selectfont,align=left}
]
  \node[item,fill=algpanel,draw=badgeAlg!55] at (0,2.65)
    {\textbf{中山引理（Nakayama lemma）。}设 $M$ 是局部环 $(R,\mathfrak{m})$ 上的\textbf{有限生成模}，若 $M = \mathfrak{m}M$ 则 $M=0$——这是交换代数与表示论中最常用的引理之一，以中山命名。};
  \node[item,fill=histpanel,draw=badgeHist!55] at (0,1.05)
    {\textbf{生成元刻画。}等价形式：若 $x_1,\dots,x_n$ 在 $M/\mathfrak{m}M$ 中生成商模，则它们在 $M$ 中生成 $M$——局部环上生成元的最小个数的核心刻画。};
  \node[item,fill=companel,draw=badgeCom!55] at (0,-0.35)
    {\textbf{广泛应用。}从代数几何的局部性质到表示论的 Cartan 矩阵，中山引理无处不在——是现代代数教科书的必备工具。};
  \node[draw=coveraccent!55,fill=graypanel,rounded corners=5pt,inner sep=6pt,text width=12.4cm,align=center] at (0,-1.7) {
    {\fontsize{7.5}{9.5}\selectfont\bfseries\color{coverdark!88} $M=\mathfrak{m}M \Rightarrow M=0$——最小生成元的核心引理。}
  };
\end{tikzpicture}
\end{center}
\end{frame}}
