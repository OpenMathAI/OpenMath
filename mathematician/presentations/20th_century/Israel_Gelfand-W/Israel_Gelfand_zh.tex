% --- 4. 泛函分析的革命 ---
\newcommand{\faslide}{%
\begin{frame}
\plainbar
\sectiontitle{泛函分析的革命：Gelfand 表示与 C*-代数 (1938–1943)}{Gelfand–Mazur 定理 · Gelfand 表示 · Gelfand–Naimark 定理 · GNS 构造}
\vspace{-0.4cm}
\begin{center}
\begin{tikzpicture}[
  item/.style={rounded corners=4pt,inner xsep=10pt,inner ysep=5pt,text width=12.4cm,font=\fontsize{7.0}{9.0}\selectfont,align=left}
]
  \node[item,fill=fapanel,draw=badgeFA!55] at (0,2.65)
    {\textbf{Gelfand–Mazur 定理 (1938)。}任何复可除 Banach 代数必同构于复数域 $\mathbb{C}$ —— 这一定理看似简单，却奠定了 Banach 代数理论近一个世纪的发展方向。它意味着 ``可除'' 与 ``复数'' 在 Banach 代数中几乎等价。};
  \node[item,fill=fapanel,draw=badgeFA!55] at (0,0.65)
    {\textbf{Gelfand 表示 (1941)。}交换 Banach 代数 $\leftrightarrow$ 紧 Hausdorff 空间上的连续函数——这是一个深刻的 ``几何化'' 对应：代数结构被完全编码在拓扑空间中。后来 Grothendieck 的概形理论正是这种精神在代数几何中的极致体现。};
  \node[item,fill=fapanel,draw=badgeFA!55] at (0,-0.80)
    {\textbf{Gelfand–Naimark 定理 (1943)。}C*-代数的基本表示定理——任何 C*-代数等距同构于某个 Hilbert 空间上的有界算子代数。{\bf GNS 构造}（Gelfand–Naimark–Segal）从正线性泛函出发构建表示——这是量子力学数学基础的核心工具，今天仍在量子信息、量子场论中广泛使用。};
  \node[draw=coveraccent!55,fill=graypanel,rounded corners=5pt,inner sep=6pt,text width=12.4cm,align=center] at (0,-2.2) {
    {\fontsize{7.5}{9.5}\selectfont\bfseries\color{coverdark!88} Banach 代数的四条基石定理中，三条以 Gelfand 命名——这就是他在泛函分析中的地位。}
  };
\end{tikzpicture}
\end{center}
\end{frame}
}
