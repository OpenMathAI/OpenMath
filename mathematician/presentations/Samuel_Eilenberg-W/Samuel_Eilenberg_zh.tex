\newcommand{\carteslide}{\begin{frame}\plainbar\sectiontitle{Cartan–Eilenberg：《同调代数》(1956)}{同调代数的圣经——定义了整整一代数学家的工具箱}

\newcommand{\singularSlide}{\begin{frame}\plainbar\sectiontitle{奇异同调与 Eilenberg–Zilber 定理}{从奇异单纯形到 Eilenberg–MacLane 谱——拓扑的其他贡献}

\node[draw=coveraccent!55, fill=graypanel, rounded corners=5pt, inner sep=6pt, text width=12.4cm, align=center] at (0,-1.85) {{\fontsize{8.0}{10.0}\selectfont\bfseries\color{coverdark!88} 七大公理 + 奇异同调 + Eilenberg–Zilber = 代数拓扑的标准工具包。}};\end{tikzpicture}\end{center}\end{frame}}
