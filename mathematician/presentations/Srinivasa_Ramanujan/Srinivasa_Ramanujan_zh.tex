\sectiontitle{Ramanujan 的 $\pi$ 公式}{\small 极快收敛的无穷级数——现代计算 $\pi$ 的核心工具}
\vspace{-0.25cm}
\begin{center}
\begin{tikzpicture}[
  item/.style={rounded corners=4pt,inner xsep=10pt,inner ysep=6pt,text width=12.4cm,font=\fontsize{7.0}{9.0}\selectfont,align=left}
]
  \node[item,fill=analysispanel,draw=badgeAnalysis!55] at (0,2.85)
    {\textbf{公式一：Ramanujan–Sato $\pi$ 级数 (1914)。}
    $\displaystyle \frac{1}{\pi} = \frac{2\sqrt{2}}{9801} \sum_{k=0}^{\infty} \frac{(4k)!\,(1103+26390k)}{(k!)^4\;396^{4k}}$
    \textbf{收敛速度令人震惊}：第一项即给出 $\pi$ 的 6 位小数；两项给出 14 位小数！\\[1pt]
    {\footnotesize\color{covermuted} 基于判别式 $d=-4\times58=-232$（类数 $h(d)=2$）。$1103+26390k$ 来自 $e^{\pi\sqrt{58}} \approx 396^4 - 104.000000177\ldots$}};
  \node[item,fill=numberpanel,draw=badgeNumber!55] at (0,1.15)
    {\textbf{公式二：惊艳 Hardy 的 $\frac{2}{\pi}$ 级数 (1913年第一封信)。}
    $\displaystyle 1 - 5\!\left(\frac{1}{2}\right)^{\!3} + 9\!\left(\frac{1\cdot 3}{2\cdot 4}\right)^{\!3} - 13\!\left(\frac{1\cdot 3\cdot 5}{2\cdot 4\cdot 6}\right)^{\!3} + \cdots = \frac{2}{\pi}$
