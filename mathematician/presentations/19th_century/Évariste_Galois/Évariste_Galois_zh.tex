\newcommand{\groupslide}{\begin{frame}\plainbar\sectiontitle{群论奠基：对称的语言}{首次使用 groupe · 正规子群 · 陪集分解}
\vspace{-0.4cm}\begin{center}\begin{tikzpicture}[item/.style={rounded corners=4pt, inner xsep=10pt, inner ysep=5pt, text width=12.4cm, font=\fontsize{7.0}{9.0}\selectfont, align=left}]

\node[item, fill=fieldpanel, draw=badgeField!40] at (0,-0.65) {\textbf{从置换到结构。}他关注的不是具体的置换，而是根之间的对称结构本身——这一视角的转换，使代数从"算方程"走向"研究结构"。};

% --- 6. 伽罗瓦理论 ---
\newcommand{\theoryslide}{\begin{frame}\plainbar\sectiontitle{伽罗瓦理论：根式可解的判据}{可解群 · 正规子群塔 · 五次及以上方程}
\vspace{-0.4cm}\begin{center}\begin{tikzpicture}[item/.style={rounded corners=4pt, inner xsep=10pt, inner ysep=5pt, text width=12.4cm, font=\fontsize{7.0}{9.0}\selectfont, align=left}]
  \node[item, fill=theorypanel, draw=badgeTheory!60] at (0,2.55) {\textbf{核心洞见。}多项式方程能否根式求解，取决于其根的置换群——即伽罗瓦群——的结构。方程可根式解，当且仅当其伽罗瓦群存在一列正规子群塔，且相邻商群皆为阿贝尔群（即伽罗瓦群"可解"）。};

\node[item, fill=redpanel, draw=revolutionred!40] at (0,-0.65) {\textbf{超越方程的视野。}这一方法后来被推广到域论、代数几何、数论等众多领域，成为现代数学中"以对称性刻画可解性"的范本。};
