% --- 6. 克莱因瓶 ---
\newcommand{\kleinbottleSlide}{%
\begin{frame}
\plainbar
\sectiontitle{克莱因瓶 (Klein Bottle)}{\faIcon{circle}\enspace 不可定向的经典 · 拓扑学的永恒标志 · 一个翻译错误成就的名字}
\vspace{-0.4cm}
\begin{center}
\begin{tikzpicture}[
  item/.style={rounded corners=4pt,inner xsep=10pt,inner ysep=5pt,text width=12.4cm,font=\fontsize{7.0}{9.0}\selectfont,align=left}
]
  \node[item,fill=topologypanel,draw=badgeTopology!55] at (0,2.65)
    {\textbf{克莱因瓶是什么？}一个单侧的（不可定向的）、无边界的闭曲面。它无法在三维欧氏空间中嵌入（必然自交），但可以在四维及更高维空间中嵌入。它可以想象为将圆柱体的一端穿过自身从内部接到另一端。};
  \node[item,fill=erlangenpanel,draw=badgeErlangen!55] at (0,0.95)
    {\textbf{"Flasche" 还是 "Fläche"？}德国原名 "Kleinsche Fläche"（Klein 曲面/面）。当翻译成英文时，"Fläche"（面）被误听/误写为 "Flasche"（瓶）。于是诞生了著名的 "Klein bottle"——一个数学史上最美丽的翻译错误。};
  \node[item,fill=legacypanel,draw=badgeLegacy!55] at (0,-0.55)
    {\textbf{Möbius 带的 3D 亲属。}克莱因瓶是三维的 Möbius 带——将两个 Möbius 带的边缘粘合起来即得。它优雅地展示了拓扑学中"定向性"这一核心概念。Klein 用它来说明：有些几何对象本质上不是局部几何所能描述的——它们有全局拓扑性质。};
  \node[draw=coveraccent!55,fill=graypanel,rounded corners=5pt,inner sep=6pt,text width=12.4cm,align=center] at (0,-2.1) {
    {\fontsize{8.0}{10.0}\selectfont\bfseries\color{coverdark!88} 一个数学概念，一个翻译错误，一段永恒传奇——Klein bottle 成为拓扑学的公共面孔。}
  };
\end{tikzpicture}
\end{center}
\end{frame}
}
